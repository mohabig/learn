import asyncio
from fastapi import FastAPI, Request
from fastapi.responses import StreamingResponse

app = FastAPI()

async def event_generator(prompt: str, request: Request):
    """Streams LLM tokens while defending against orphaned compute leaks."""
    try:
        # Simulate upstream token generator
        for i in range(50):
            if await request.is_disconnected():
                print("Client disconnected early! Halting upstream inference.")
                break
            yield f"data: {{\"chunk\": {i}}}\n\n"
            await asyncio.sleep(0.04)
    finally:
        # Guarantee resources, VRAM slots, and DB connections are released!
        print("Cleaning up generator resources.")

@app.post("/stream")
async def stream_tokens(request: Request):
    return StreamingResponse(
        event_generator("sample prompt", request),
        media_type="text/event-stream",
        headers={"X-Accel-Buffering": "no", "Cache-Control": "no-cache"}
    )
